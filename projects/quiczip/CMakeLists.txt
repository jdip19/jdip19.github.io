cmake_minimum_required(VERSION 3.20)
project(QuickZip VERSION 1.0.0 LANGUAGES CXX)

include(FetchContent)
set(WEBP_BUILD_ANIM_UTILS OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_CWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_DWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_GIF2WEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_IMG2WEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_VWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_WEBPINFO OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(webp GIT_REPOSITORY https://chromium.googlesource.com/webm/libwebp.git GIT_TAG v1.5.0)
FetchContent_MakeAvailable(webp)

add_executable(QuickZip WIN32 src/main.cpp resources/resource.rc)
target_compile_features(QuickZip PRIVATE cxx_std_20)
target_compile_definitions(QuickZip PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(QuickZip PRIVATE webp user32 shell32 ole32 oleaut32 shlwapi windowscodecs comctl32 advapi32)
if(MINGW)
  target_link_options(QuickZip PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
set_target_properties(QuickZip PROPERTIES OUTPUT_NAME "QuickZip")

include(GNUInstallDirs)
install(TARGETS QuickZip RUNTIME DESTINATION .)
